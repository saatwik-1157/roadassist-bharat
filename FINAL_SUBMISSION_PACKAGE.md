## Creating the ZIP — Windows PowerShell

```powershell
cd C:\Users\Asus\project_nirdhan
$src = "roadassist-bharat"
$dst = "$env:USERPROFILE\Desktop\RoadAssist-Bharat-Final"
$exclude = @('node_modules','dist','build','.gradle','.git','uploads','assets')

robocopy $src $dst /E /XD $exclude /XF '*.log' '.env' | Out-Null
Compress-Archive -Path $dst -DestinationPath "$dst.zip" -Force
"{0:N1} MB" -f ((Get-Item "$dst.zip").Length / 1MB)
```

## Verify the ZIP

```powershell
Add-Type -A System.IO.Compression.FileSystem
$z = [IO.Compression.ZipFile]::OpenRead("$env:USERPROFILE\Desktop\RoadAssist-Bharat-Final.zip")
"entries: $($z.Entries.Count)"
foreach ($m in 'README.md','FINAL.pdf','seed.ts','app.html','server.ts') {
  $hit = $z.Entries | Where-Object { $_.FullName -like "*$m" } | Select-Object -First 1
  "{0,-14} {1}" -f $m, $(if ($hit) { "OK" } else { "MISSING" })
}
$bad = $z.Entries | Where-Object { $_.FullName -match 'node_modules|\.env$|\.log$' }
"unwanted:  $($bad.Count)"      # must be 0
$z.Dispose()
```
